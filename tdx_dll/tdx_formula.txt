{公式名称:ZEN}
{公式类型:主图指标}

{=============================================================================}
{                           禅中看缠 (ZEN PLUGIN) 主图公式                    }
{        默认模式 MODE: 9004 (仅拓展，缺口成笔/新笔/扩张/跨界均关闭)        }
{=============================================================================}

{--- 1. 分析模式 ---}
{ PMODE=1 逐根推送(默认) / PMODE=0 整体推送 }
{ 在公式参数中新增 PMODE，默认值 1，通过 ALT+T 动态切换 }
PSET:=TDXDLL2(IF(PMODE<0.5,801,800),H,C,L);
CFG:=TDXDLL2(MODE,H,C,L);

{--- 2. 基础价格单元 ---}
{ BASE 不用 CONST，逐根取 C，使偏移量随当前查看区域价格自适应 }
BASE:=C;
UNIT_RANGE:=(HHV(H,120)-LLV(L,120))/100;
UNIT_PRICE:=IF(BASE<5,0.005,IF(BASE<10,0.01,IF(BASE<30,0.025,IF(BASE<60,0.05,
            IF(BASE<100,0.1,IF(BASE<200,0.2,IF(BASE<400,0.4,IF(BASE<1000,1.0,BASE*0.001))))))));
UNIT:=MAX(UNIT_RANGE,UNIT_PRICE);

{--- 3. 对齐参数 ---}
GAP:=UNIT*2.5;
SH:=UNIT*0.8;
{ ICON 像素中心比文字偏低，不同 ICON 图形偏差不同，分别补偿 }
{ 放大时用 BASE*0.0001 保底，缩小时用 UNIT_RANGE 主导 }
ICON_ADJ_BASE:=MAX(UNIT_RANGE,BASE*0.0001);
ICON_ADJ_ARROW:=ICON_ADJ_BASE*0.3;
ICON_ADJ_HAND:=ICON_ADJ_BASE*0.2;
ICON_ADJ_TRI:=ICON_ADJ_BASE*1.0;

{--- 4. 笔 ---}
R_BI:=TDXDLL2(110,H,C,L);
R_BI_PEND:=TDXDLL2(130,H,C,L);
BI:IF(R_BI>0.01 AND R_BI<999999 AND R_BI_PEND<=0.5,R_BI,DRAWNULL),COLORMAGENTA,LINETHICK1;
BI_P:IF(R_BI>0.01 AND R_BI<999999 AND R_BI_PEND>=0.5,R_BI,DRAWNULL),COLORMAGENTA,LINETHICK1,DOTLINE;
{ BI_Y: 不受掩码影响的笔价格，用于买卖点/背驰的 Y 坐标计算 }
BI_Y:=IF(R_BI>0.01 AND R_BI<999999,R_BI,DRAWNULL);

{--- 5. 笔端点价格 (代码部分的大括号移除即可显示价格，目前文字太多隐藏起来) ---}
{ 顶分型端点 = 向上笔终点 / 向下笔起点：价格显示在端点上方 }
{BI_TOP:=R_BI>0.01 AND R_BI<999999 AND R_BI>REF(R_BI,1) AND R_BI>REFX(R_BI,1);}
{ 底分型端点 = 向上笔起点 / 向下笔终点：价格显示在端点下方 }
{BI_BOT:=R_BI>0.01 AND R_BI<999999 AND R_BI<REF(R_BI,1) AND R_BI<REFX(R_BI,1);}
{ 小数位数统一 2 位 }
{ 偏移量取 UNIT 的 60%，让价格标签与端点保持适当间距 }
{BI_OFF:=UNIT*0.6;}
{DRAWTEXT(BI_TOP, R_BI+BI_OFF, VAR2STR(R_BI, 2)),COLORMAGENTA;}
{DRAWTEXT(BI_BOT, R_BI-BI_OFF, VAR2STR(R_BI, 2)),COLORMAGENTA;}

{--- 6. 线段 ---}
R_XD:=TDXDLL2(120,H,C,L);
R_XD_PEND:=TDXDLL2(131,H,C,L);
XD:IF(R_XD>0.01 AND R_XD<999999 AND R_XD_PEND<=0.5,R_XD,DRAWNULL),COLORYELLOW,LINETHICK2;
XD_P:IF(R_XD>0.01 AND R_XD<999999 AND R_XD_PEND>=0.5,R_XD,DRAWNULL),COLORYELLOW,LINETHICK2,CIRCLEDOT;
{ XD_Y: 不受掩码影响的线段价格，用于买卖点/背驰的 Y 坐标计算 }
XD_Y:=IF(R_XD>0.01 AND R_XD<999999,R_XD,DRAWNULL);

{--- 7. 笔中枢 ---}
BZG:=TDXDLL2(210,H,C,L); BZD:=TDXDLL2(211,H,C,L);
STICKLINE(BZG>0.01,BZG,BZD,0,-1),COLORMAGENTA;
BI_ZD:IF(BZD>0.01,BZD,DRAWNULL),COLORMAGENTA,DOTLINE;
BI_ZG:IF(BZG>0.01,BZG,DRAWNULL),COLORMAGENTA,DOTLINE;

{--- 8. 线段中枢 ---}
XZG:=TDXDLL2(220,H,C,L); XZD:=TDXDLL2(221,H,C,L);
XZ_S:=XZG>0.01 AND (REF(XZG,1)<=0.01 OR REF(XZG,1)!=XZG);
XZ_E:=XZG>0.01 AND (REFX(XZG,1)<=0.01 OR REFX(XZG,1)!=XZG OR ISLASTBAR);
STICKLINE(XZ_S OR XZ_E,XZG,XZD,0,-1),COLORYELLOW,LINETHICK1;
XD_ZD:IF(XZD>0.01,XZD,DRAWNULL),COLORYELLOW,DOTLINE;
XD_ZG:IF(XZG>0.01,XZG,DRAWNULL),COLORYELLOW,DOTLINE;

{--- 9. 笔信号解析 ---}
SIG:=TDXDLL2(218,H,C,L);
TP:=MOD(ABS(SIG),100);
BC:=INTPART(MOD(ABS(SIG),1000)/100);
IS_BUY:=SIG>0 AND ((TP>=1 AND TP<=3) OR TP=23);
IS_SELL:=SIG<0 AND ((TP>=1 AND TP<=3) OR TP=23);

{--- 10. 笔中枢买卖点 ---}
B_BUY_Y:=BI_Y-GAP; B_SELL_Y:=BI_Y+GAP;

DRAWICON(IS_BUY,B_BUY_Y+SH+ICON_ADJ_ARROW,1);
DRAWTEXT(IS_BUY AND TP=1, B_BUY_Y-SH,'  1B'),COLORWHITE;
DRAWTEXT(IS_BUY AND TP=2, B_BUY_Y-SH,'  2B'),COLORWHITE;
DRAWTEXT(IS_BUY AND TP=3, B_BUY_Y-SH,'  3B'),COLORWHITE;
DRAWTEXT(IS_BUY AND TP=23,B_BUY_Y-SH,'  23B'),COLORWHITE;

DRAWICON(IS_SELL,B_SELL_Y+SH+ICON_ADJ_ARROW,2);
DRAWTEXT(IS_SELL AND TP=1, B_SELL_Y-SH,'  1S'),COLORGREEN;
DRAWTEXT(IS_SELL AND TP=2, B_SELL_Y-SH,'  2S'),COLORGREEN;
DRAWTEXT(IS_SELL AND TP=3, B_SELL_Y-SH,'  3S'),COLORGREEN;
DRAWTEXT(IS_SELL AND TP=23,B_SELL_Y-SH,'  23S'),COLORGREEN;

{--- 11. 笔中枢趋势背驰 ---}
STR:=IF(BC=1,'(弱)',IF(BC=2,'(中)',IF(BC=3,'(强)','')));
D_BUY_Y:=BI_Y-GAP*2.0; D_SELL_Y:=BI_Y+GAP*2.0;

DRAWICON(SIG>0 AND BC>0 AND TP>0,D_BUY_Y+SH+ICON_ADJ_HAND,11);
DRAWTEXT(SIG>0 AND BC>0 AND TP>0,D_BUY_Y-SH,VARCAT('   底背驰',STR)),COLORWHITE;
DRAWICON(SIG<0 AND BC>0 AND TP>0,D_SELL_Y+SH+ICON_ADJ_HAND,12);
DRAWTEXT(SIG<0 AND BC>0 AND TP>0,D_SELL_Y-SH,VARCAT('   顶背驰',STR)),COLORGREEN;

{--- 12. 笔中枢盘整背驰 ---}
BZP:=ABS(SIG)>=100 AND TP=0;
BZP_BUY:=BZP AND SIG>0;
BZP_SELL:=BZP AND SIG<0;
DRAWICON(BZP_BUY AND BC>0,D_BUY_Y+SH+ICON_ADJ_HAND,11);
DRAWTEXT(BZP_BUY AND BC>0,D_BUY_Y-SH,VARCAT('   盘整背驰',STR)),COLORYELLOW;
DRAWICON(BZP_SELL AND BC>0,D_SELL_Y+SH+ICON_ADJ_HAND,12);
DRAWTEXT(BZP_SELL AND BC>0,D_SELL_Y-SH,VARCAT('   盘整背驰',STR)),COLORYELLOW;

{--- 13. 线段中枢信号解析 ---}
SIG_XD:=TDXDLL2(228,H,C,L);
TP_XD:=MOD(ABS(SIG_XD),100);
BC_XD:=INTPART(MOD(ABS(SIG_XD),1000)/100);
IS_BUY_XD:=SIG_XD>0 AND ((TP_XD>=1 AND TP_XD<=3) OR TP_XD=23);
IS_SELL_XD:=SIG_XD<0 AND ((TP_XD>=1 AND TP_XD<=3) OR TP_XD=23);

{--- 14. 线段中枢买卖点 ---}
X_BUY_Y:=XD_Y-GAP*1.5; X_SELL_Y:=XD_Y+GAP*1.5;

DRAWICON(IS_BUY_XD,X_BUY_Y+SH+ICON_ADJ_ARROW,1);
DRAWTEXT(IS_BUY_XD AND TP_XD=1, X_BUY_Y-SH,'  1B-XD'),COLORWHITE;
DRAWTEXT(IS_BUY_XD AND TP_XD=2, X_BUY_Y-SH,'  2B-XD'),COLORWHITE;
DRAWTEXT(IS_BUY_XD AND TP_XD=3, X_BUY_Y-SH,'  3B-XD'),COLORWHITE;
DRAWTEXT(IS_BUY_XD AND TP_XD=23,X_BUY_Y-SH,'  23B-XD'),COLORWHITE;

DRAWICON(IS_SELL_XD,X_SELL_Y+SH+ICON_ADJ_ARROW,2);
DRAWTEXT(IS_SELL_XD AND TP_XD=1, X_SELL_Y-SH,'  1S-XD'),COLORGREEN;
DRAWTEXT(IS_SELL_XD AND TP_XD=2, X_SELL_Y-SH,'  2S-XD'),COLORGREEN;
DRAWTEXT(IS_SELL_XD AND TP_XD=3, X_SELL_Y-SH,'  3S-XD'),COLORGREEN;
DRAWTEXT(IS_SELL_XD AND TP_XD=23,X_SELL_Y-SH,'  23S-XD'),COLORGREEN;

{--- 15. 线段中枢趋势背驰 ---}
STR_XD:=IF(BC_XD=1,'(弱)',IF(BC_XD=2,'(中)',IF(BC_XD=3,'(强)','')));
DX_BUY_Y:=XD_Y-GAP*2.5; DX_SELL_Y:=XD_Y+GAP*2.5;

DRAWICON(SIG_XD>0 AND BC_XD>0 AND TP_XD>0,DX_BUY_Y+SH+ICON_ADJ_HAND,11);
DRAWTEXT(SIG_XD>0 AND BC_XD>0 AND TP_XD>0,DX_BUY_Y-SH,VARCAT('   底背驰-XD',STR_XD)),COLORWHITE;
DRAWICON(SIG_XD<0 AND BC_XD>0 AND TP_XD>0,DX_SELL_Y+SH+ICON_ADJ_HAND,12);
DRAWTEXT(SIG_XD<0 AND BC_XD>0 AND TP_XD>0,DX_SELL_Y-SH,VARCAT('   顶背驰-XD',STR_XD)),COLORGREEN;

{--- 16. 线段中枢盘整背驰 ---}
BZP_XD:=ABS(SIG_XD)>=100 AND TP_XD=0;
BZP_XD_BUY:=BZP_XD AND SIG_XD>0;
BZP_XD_SELL:=BZP_XD AND SIG_XD<0;
DRAWICON(BZP_XD_BUY AND BC_XD>0,DX_BUY_Y+SH+ICON_ADJ_HAND,11);
DRAWTEXT(BZP_XD_BUY AND BC_XD>0,DX_BUY_Y-SH,VARCAT('   盘整背驰-XD',STR_XD)),COLORYELLOW;
DRAWICON(BZP_XD_SELL AND BC_XD>0,DX_SELL_Y+SH+ICON_ADJ_HAND,12);
DRAWTEXT(BZP_XD_SELL AND BC_XD>0,DX_SELL_Y-SH,VARCAT('   盘整背驰-XD',STR_XD)),COLORYELLOW;

{--- 17. MACD背离（独立于缠论，仅供参考，不要跟缠论的背驰混淆） ---}
M_DIV:=TDXDLL2(301,H,C,L);
BC_M:=MOD(ABS(M_DIV),10);
STR_M:=IF(BC_M=1,'(弱)',IF(BC_M=2,'(中)',IF(BC_M=3,'(强)','')));
M_BUY_Y:=L-GAP*3.0; M_SELL_Y:=H+GAP*3.0;

DRAWICON(M_DIV>=10,M_BUY_Y+SH+ICON_ADJ_TRI,38);
DRAWTEXT(M_DIV>=10,M_BUY_Y-SH,VARCAT('   MACD底背',STR_M)),COLORWHITE;
DRAWICON(M_DIV<=-10,M_SELL_Y+SH+ICON_ADJ_TRI,39);
DRAWTEXT(M_DIV<=-10,M_SELL_Y-SH,VARCAT('   MACD顶背',STR_M)),COLORGREEN;
